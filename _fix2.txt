24|static inline int wrap_delta_i(int a, int b, int dim) {
25|    int d = (a - b) % dim;
26|    if (d > dim / 2) d -= dim;
27|    if (d < -dim / 2) d += dim;
28|    return d;

31|static inline float wrap_delta_f_local(float a, float b, float extent) {
32|    float d = std::fmod(a - b, extent);
33|    if (d > extent * 0.5f) d -= extent;
34|    if (d < -extent * 0.5f) d += extent;
35|    return d;

39|static inline std::uint32_t spatial_hash(int x, int y, int z, std::uint32_t seed) {
40|    std::uint32_t h = static_cast<std::uint32_t>(x) * 73856093u ^
41|                      static_cast<std::uint32_t>(y) * 19349663u ^
42|                      static_cast<std::uint32_t>(z) * 83492791u ^ seed;
43|    h = (h ^ (h >> 16)) * 0x45d9f3bu;
44|    h = (h ^ (h >> 16)) * 0x45d9f3bu;
45|    return h ^ (h >> 16);

94|bool check_projectile_prop_hits(Registry& reg, const vec3& projPos, const vec3& projVel,
95|                                float projHitRadius, EventBus& bus)
96|{
97|    const float radiusSq = projHitRadius * projHitRadius;
98|    const int pcx = wrap_macro_local(static_cast<int>(projPos.x / kCellSize));
99|    const int pcy = wrap_macro_local(static_cast<int>(projPos.y / kCellSize));
100|    const int pcz = wrap_macro_local(static_cast<int>(projPos.z / kCellSize));
101|
102|    Entity hitEntity = entt::null;
103|    PropFallMode hitMode = PropFallMode::SimpleFall;
104|    vec3 hitPos{0.0f, 0.0f, 0.0f};
105|    vec3 hitColor{0.8f, 0.8f, 0.8f};
106|
107|    auto view = reg.view<Transform, SubVoxelAnchor, PropFallMode>();
108|    for (auto entity : view) {
109|        const auto& anchor = view.get<SubVoxelAnchor>(entity);
110|
111|        if (std::abs(wrap_delta_i(anchor.cx, pcx, kMacroDimLocal)) > 1 ||
112|            std::abs(wrap_delta_i(anchor.cy, pcy, kMacroDimLocal)) > 1 ||
113|            std::abs(wrap_delta_i(anchor.cz, pcz, kMacroDimLocal)) > 1)
114|        {
115|            continue;
116|        }
117|
118|        const auto& tr = view.get<Transform>(entity);
119|        const float dx = wrap_delta_f_local(projPos.x, tr.pos.x, kWorldExtentLocal);
120|        const float dy = wrap_delta_f_local(projPos.y, tr.pos.y, kWorldExtentLocal);
121|        const float dz = wrap_delta_f_local(projPos.z, tr.pos.z, kWorldExtentLocal);
122|
123|        if (dx * dx + dy * dy + dz * dz <= radiusSq) {
124|            hitEntity = entity;
125|            hitMode = view.get<PropFallMode>(entity);
126|            hitPos = tr.pos;
127|            hitColor = reg.all_of<Renderable>(entity) ? reg.get<Renderable>(entity).color : vec3{0.8f, 0.8f, 0.8f};
128|            break;
129|        }
130|    }
131|
132|    if (hitEntity != entt::null) {
133|        vec3 impulse = normalize(projVel) * 3.0f + vec3{0.0f, 1.0f, 0.0f};
134|        detach_single_prop(reg, hitEntity, hitMode, impulse, hitPos, hitColor, 0, bus);
135|        return true;
136|    }
137|    return false;

140|void anchor_validate_step(Registry& reg, const World& world, EventBus& bus,
141|                         const std::vector<std::uint32_t>& dirtyCells)
142|{
143|    if (dirtyCells.empty()) return;
144|
145|    // dirtyCells are flat macro_index keys (CarveResult / DoorSet contract).
146|    static thread_local std::unordered_set<std::uint32_t> dirtySet;
147|    dirtySet.clear();
148|    dirtySet.insert(dirtyCells.begin(), dirtyCells.end());
149|
150|    static thread_local std::vector<PendingDetachedProp> detached;
151|    detached.clear();
152|
153|    auto view = reg.view<Transform, SubVoxelAnchor, PropFallMode>();
154|    for (auto entity : view) {
155|        const auto& anchor = view.get<SubVoxelAnchor>(entity);
156|
157|        const int cx = wrap_macro_local(anchor.cx);
158|        const int cy = wrap_macro_local(anchor.cy);
159|        const int cz = wrap_macro_local(anchor.cz);
160|        const std::uint32_t key =
161|            static_cast<std::uint32_t>(macro_index(cx, cy, cz));
162|
163|        if (!dirtySet.contains(key)) continue;
164|
165|        // Anchor support lost when the sub-voxel is no longer solid.
166|        if (!world.grid().solid(cx, cy, cz, anchor.subX, anchor.subY, anchor.subZ)) {
167|            const auto& tr = view.get<Transform>(entity);
168|            vec3 col = reg.all_of<Renderable>(entity)
169|                           ? reg.get<Renderable>(entity).color
170|                           : vec3{0.8f, 0.8f, 0.8f};
171|            detached.push_back({entity, view.get<PropFallMode>(entity), tr.pos,
172|                                vec3{0.0f, 1.0f, 0.0f}, col, 0});
173|        }
174|    }
175|
176|    for (const auto& item : detached) {
177|        detach_single_prop(reg, item.entity, item.mode, item.impulse, item.pos,
178|                           item.color, item.meshKind, bus);
179|    }

226|std::uint32_t seed_wall_interactables(Registry& reg, const World& world,
227|                                      LayerId layer, std::uint32_t seed)
228|{
229|    const MacroGrid& grid = world.grid();
230|    std::uint32_t count = 0;
231|    constexpr float kCell = kCellSize;
232|
233|    // Mirror PropPlacer::populate wall-device branch (wsel bands 15-25 shield,
234|    // 25-35 terminal). Floor support = solidBelow on Y (same convention as
235|    // prop_placer). Anchor sub-voxels point into the solid floor cell so
236|    // spawn_prop's solid() check and anchor_validate_step stay honest.
237|    for (int z = 0; z < kMacroDimLocal; ++z) {
238|        for (int y = 0; y < kMacroDimLocal; ++y) {
239|            for (int x = 0; x < kMacroDimLocal; ++x) {
240|                if (grid.cell(x, y, z) != kCellAir) continue;
241|
242|                const CellType below = grid.cell(x, y - 1, z);
243|                if (!is_solid_cell(below)) continue;
244|
245|                const bool solidWest  = is_solid_cell(grid.cell(x - 1, y, z));
246|                const bool solidEast  = is_solid_cell(grid.cell(x + 1, y, z));
247|                const bool solidNorth = is_solid_cell(grid.cell(x, y, z + 1));
248|                const bool solidSouth = is_solid_cell(grid.cell(x, y, z - 1));
249|                if (!(solidWest || solidEast || solidNorth || solidSouth)) continue;
250|
251|                const std::uint32_t rngWall = spatial_hash(x, y, z, seed ^ kSaltWall);
252|                const std::uint32_t wsel = rngWall % 100;
253|
254|                Interactable::Kind kind;
255|                float yOff;
256|                vec3 color;
257|                std::uint8_t face = 1; // wall
258|                if (wsel >= 15 && wsel < 25) {
259|                    kind  = Interactable::Kind::ElectricalShield;
260|                    yOff  = 0.40f;
261|                    color = {0.18f, 0.20f, 0.22f};
262|                } else if (wsel >= 25 && wsel < 35) {
263|                    kind  = Interactable::Kind::Terminal;
264|                    yOff  = 0.0f;
265|                    color = {0.32f, 0.35f, 0.38f};
266|                } else {
267|                    continue; // radiator / empty — no Interactable
268|                }
269|
270|                const float wx = static_cast<float>(x) * kCell;
271|                const float wy = static_cast<float>(y) * kCell + yOff;
272|                const float wz = static_cast<float>(z) * kCell;
273|
274|                // Anchor into the solid floor cell under the air cell (Y-1).
275|                SubVoxelAnchor anchor;
276|                anchor.cx   = x;
277|                anchor.cy   = wrap_macro_local(y - 1);
278|                anchor.cz   = z;
279|                anchor.subX = 4;
280|                anchor.subY = 7; // top of floor cell
281|                anchor.subZ = 4;
282|                anchor.face = face;
283|
284|                Entity e = spawn_prop(reg, world, vec3{wx, wy, wz}, anchor, kind,
285|                                      PropFallMode::SimpleFall, color, 0, layer);
286|                if (e != entt::null) ++count;
287|            }
288|        }
289|    }
290|    return count;

310|InteractionHit find_nearest_interactable(const Registry& reg, Entity player,
311|                                         Interactable::Kind kind, float reachM)
312|{
313|    InteractionHit hit{};
314|    if (!reg.valid(player) || !reg.all_of<Transform>(player)) return hit;
315|
316|    const auto& ptr = reg.get<Transform>(player);
317|    const LayerId layer = ptr.layer;
318|    const vec3 ppos = ptr.pos;
319|    const float reachSq = reachM * reachM;
320|    float best = reachSq;
321|
322|    auto view = reg.view<const Transform, const Interactable>();
323|    for (auto e : view) {
324|        const auto& tr = view.get<const Transform>(e);
325|        if (tr.layer != layer) continue;
326|        const auto& ia = view.get<const Interactable>(e);
327|        if (!ia.active || ia.kind != kind) continue;
328|
329|        const float dx = wrap_delta_f_local(ppos.x, tr.pos.x, kWorldExtentLocal);
330|        const float dy = ppos.y - tr.pos.y;
331|        const float dz = wrap_delta_f_local(ppos.z, tr.pos.z, kWorldExtentLocal);
332|        const float d2 = dx * dx + dy * dy + dz * dz;
333|        if (d2 < best) {
334|            best = d2;
335|            hit.entity = e;
336|            hit.pos = tr.pos;
337|            hit.distSq = d2;
338|            hit.hit = true;
339|        }
340|    }
341|    return hit;

356|void prop_ragdoll_step(Registry& reg, float dt)
357|{
358|    (void)dt;
359|    // Angular integration is owned by physics_step (core). Here we only
360|    // settle spin on grounded DynamicBodyTag props so tumbling debris stops
361|    // spinning after landing ([jirnyak.md] §18 prop_ragdoll).
362|    auto view = reg.view<DynamicBodyTag, AngularVelocity, GravityAffected>();
363|    for (auto e : view) {
364|        const auto& g = view.get<GravityAffected>(e);
365|        if (!g.grounded) continue;
366|        auto& ang = view.get<AngularVelocity>(e);
367|        ang.w.x *= 0.85f;
368|        ang.w.y *= 0.85f;
369|        ang.w.z *= 0.85f;
370|        const float w2 = ang.w.x * ang.w.x + ang.w.y * ang.w.y + ang.w.z * ang.w.z;
371|        if (w2 < 1e-4f) {
372|            ang.w = vec3{0.0f, 0.0f, 0.0f};
373|        }
374|    }

// Cell material ids — the shared vocabulary between the generators that WRITE
// cells and the render pass that gives them a colour.
//
// `CellType` is an opaque `uint16` to the engine ([voxels.md]); these constants
// are the game's meanings for it. They live in `world/` rather than `game/` for
// one reason: the render pass needs them and render must never include `game/`
// ([ARCHITECTURE.md] layering). Nothing here is behaviour — a material id is a
// label, and the only code that branches on it is the colour table in
// `render/cube_pass.cpp`.
//
// Adding a material is one id here plus one row there, per the data-driven rule in
// [AGENTS.md] — never an `if` in the generator.
#pragma once

#include "world/macro_grid.h"  // CellType, kCellAir

namespace giga {

// --- The original demo vocabulary -------------------------------------------
// Used by the maze test bed (`app/worldgen.cpp`). Kept at their historical ids so
// existing saves and the maze mode keep rendering as they did.
inline constexpr CellType kMatConcrete = 1;  // generic grey wall / stone
inline constexpr CellType kMatSoil = 2;
inline constexpr CellType kMatWaterMark = 3;
inline constexpr CellType kMatSlabTan = 4;   // generic tan slab
inline constexpr CellType kMatHubPad = 7;    // nav / fast-travel pad (cyan)

// The bank. Takes the free id 5 rather than extending the table, so kMatCount and
// every count-drift check stay exactly where they were.
//
// It needs its own material because reusing kMatHubPad DID NOT WORK, and the way it
// failed is worth recording: the hub pads are stamped onto the slab at the four
// lattice z-levels {16, 48, 80, 112}, while a body walks at cell z=1. So the pad was
// always 15 cells (30 m) above the player's feet and the extraction check could
// never once return true. Nothing would have crashed, nothing would have warned —
// banking would simply never have happened. Found by reading the generator, not by
// playing, which is the only way this class of bug gets found.
inline constexpr CellType kMatExtract = 5;   // extraction pad / bank (emerald)

// Doors ([game/door.h]) — the shut leaf AND the permanent frame around every
// doorway, so a doorway is findable at range whether or not its door is closed.
//
// Takes id 6, the one id the tables already carried and no generator wrote, for a
// harder reason than tidiness: kMatCount must stay at 16. The albedo table in
// render/cube_pass.cpp is sized from it and would simply grow, but
// shaders/material_surface.glsl is GENERATED with `uint[16]` literals and
// cube.frag clamps the incoming id with `min(mat, kMatSurfaceCount - 1u)`. So an
// id 16 would silently render with the RUBBLE surface family — right colour,
// wrong material character, no warning on either host. Reusing 6 keeps every id
// inside the generated tables and leaves the CSV drift gate untouched.
//
// One id and not two: an OPEN door is air, so there is no leaf to draw, and what
// makes the doorway legible in both states is the frame — the two jambs plus the
// lintel, recoloured in place over cells that were already solid wall. Shut reads
// as the frame filled in, open as the frame around a hole.
inline constexpr CellType kMatDoor = 6;      // door leaf + frame (painted steel)

// --- Khrushchevka materials, one pair per FloorKind -------------------------
// Before these existed every floor kind wrote kMatConcrete + kMatSlabTan, so an
// Industrial pillar plate and a Residential warren rendered in identical grey and
// tan — the maze demo's palette, on a Soviet apartment block.
//
// Albedos for the metal/industrial half are **measured** off real 2K photographs
// (Poly Haven, CC0; see data/materials.csv for the per-material source URL and the
// measured luminance variance). The residential half is authored, because the pack
// contains no wallpaper, parquet, plaster or linoleum — that is the honest split:
// harvest where a real material exists, author where none does.
inline constexpr CellType kMatPlaster = 8;      // Residential wall  (authored)
inline constexpr CellType kMatParquet = 9;      // Residential slab  (authored)
inline constexpr CellType kMatShopShutter = 10; // Commercial wall   (measured)
inline constexpr CellType kMatLino = 11;        // Commercial slab   (measured)
inline constexpr CellType kMatFactoryWall = 12; // Industrial wall   (measured)
inline constexpr CellType kMatTread = 13;       // Industrial slab   (measured)
inline constexpr CellType kMatRust = 14;        // Derelict wall     (measured)
inline constexpr CellType kMatRubble = 15;      // Derelict slab     (measured)

// --- Environmental hazards --------------------------------------------------
inline constexpr CellType kMatElectricGrate = 16; // Electrical floor grate
inline constexpr CellType kMatAcidPool     = 17; // Acid pool
inline constexpr CellType kMatFireCell     = 18; // Fire cell / burning floor

// One past the last id in use. The colour table is sized from this, so adding a
// material without extending the table fails the build rather than rendering as
// an unremarkable default.
inline constexpr CellType kMatCount = 19;

} // namespace giga
