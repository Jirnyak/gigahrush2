268: // 8 shifts per byte over a ~3.6 KB save is a few microseconds — a lookup table would be

271: std::uint32_t crc32(const std::uint8_t* p, std::size_t n) {
272:     std::uint32_t c = 0xFFFFFFFFu;
273:     for (std::size_t i = 0; i < n; ++i) {
274:         c ^= static_cast<std::uint32_t>(p[i]);
275:         for (int k = 0; k < 8; ++k)
276:             c = (c >> 1) ^ (0xEDB88320u & (0u - (c & 1u)));

1: // Save / load — the format, and the two ways it is allowed to fail.
2: //
3: // Included into game_test.cpp, so it uses that file's CHECK macro and its
4: // `using namespace giga::game`. Everything except the single entry point
5: // `test_saveload_all()` lives in `namespace saveload_test`.
6: //
7: // Two load-bearing tests, and they are testing opposite things:
8: //
9: //   * `round_trip()` proves the format is lossless. It fills EVERY field with a
10: //     distinct value first, because a round-trip over zeroes passes even when the
11: //     serializer drops half the struct.
12: //   * `weak_check_vs_strong_check()` proves the format refuses a save it would
13: //     otherwise misread. That is the whole reason this file exists: `ItemId` is a row
14: //     index into an alphabetically-sorted CSV, so a reordered table turns a valid save
15: //     into a valid save that means something else. It re-derives the fingerprint
16: //     independently, then shows a REORDER at constant row count — the case
17: //     `kItemCount` cannot see — is rejected.
18: //
19: // A note on "byte-for-byte", because the phrase is easy to get wrong here. The
20: // assertions compare the SERIALIZED BYTES, not the structs. `Contract` carries two bytes
21: // of implicit padding between `subject` and `target` and `RunLedger` seven at its tail,
22: // and the standard says nothing about what aggregate initialization puts in them — so a
23: // `memcmp` of two of those structs is a test that could fail for a reason that is not a
24: // bug. The wire has no padding at all (every field is written individually), so a
25: // byte-for-byte comparison there is exact and means what it says. `Needs` and
26: // `Inventory` have no implicit padding — 8x4+1+3 and 64x4 — so those two ARE compared
27: // with memcmp.

57: void poke_u32(std::vector<std::uint8_t>& buf, std::size_t at, std::uint32_t v) {
58:     for (int i = 0; i < 4; ++i)
59:         buf[at + static_cast<std::size_t>(i)] =
60:             static_cast<std::uint8_t>((v >> (i * 8)) & 0xFFu);

63: // CRC-32 again, written independently. Two jobs: it lets a test re-checksum a payload it
64: // edited on purpose, and — pinned below against the standard's own check value — it
65: // proves the production checksum really is CRC-32 rather than "a hash that agrees with
66: // itself", which is all a same-implementation comparison could ever show.
67: std::uint32_t crc32_over(const std::uint8_t* p, std::size_t n) {
68:     std::uint32_t c = 0xFFFFFFFFu;
69:     for (std::size_t i = 0; i < n; ++i) {
70:         c ^= static_cast<std::uint32_t>(p[i]);
71:         for (int k = 0; k < 8; ++k)
72:             c = (c >> 1) ^ (0xEDB88320u & (0u - (c & 1u)));
73:     }
74:     return ~c;

207: void wire_layout() {
208:     // The format's footprint is arithmetic, not a measurement — a save whose length
209:     // depends on the compiler is a save that cannot cross hosts.
210:     // Derived from the serializers, not measured from a run: 33 ledger + 79 book
211:     // (3 x 21 + 16) + 304 player (33 needs + 256 inventory + 12 + 3) + 308 quest log
212:     // (20 rows x 14 + 8 earned + 5 x 4 counters) = 724, plus the fixed 36-byte
213:     // faction matrix and the 64-byte header (48 v1 + 8 v2 quests + 8 v6 blobs).
214:     static_assert(kSaveHeaderWire == 64);
215:     static_assert(kSaveFixedWire == 724);
216:     static_assert(kFactionWire == 36);
217:     static_assert(save_bytes_for(0) == 824);
218:     static_assert(save_bytes_for(3) == 824 + 15);
219:     static_assert(save_bytes_for(3, 100, 50) == 824 + 15 + 150);
220: 
221:     std::vector<std::uint8_t> bytes;
222:     SaveState empty;
223:     save_write(empty, bytes);
224:     CHECK(bytes.size() == save_bytes_for(0));
225: 
226:     const SaveState st = busy_run();

228:     CHECK(bytes.size() == save_bytes_for(3));
229:     // 839 B for a full run with three emptied crates and no macro blobs (those are
230:     // variable-size and pinned by macro_world_round_trips). GEOMETRY lives in the
231:     // per-floor files ([save.h] modular layout), never here.
232:     CHECK(bytes.size() == 839);
233: 
234:     // The magic is readable in a hex dump: 'G' 'H' '2' 'S'.
235:     CHECK(bytes[0] == 'G');
236:     CHECK(bytes[1] == 'H');

360: void craft_write(const CraftingState& st, std::uint8_t* out) {
361:     if (!out) return;
362:     std::uint8_t* p = out;
363:     for (std::size_t w = 0; w < kCraftKnownWords; ++w) {
364:         put_u32(p, static_cast<std::uint32_t>(st.known[w] & 0xffffffffull));
365:         put_u32(p, static_cast<std::uint32_t>(st.known[w] >> 32));

371: bool craft_read(const std::uint8_t* in, std::size_t n, CraftingState& st) {
372:     if (!in || n < kCraftingWire) return false;   // untouched on refusal
373:     CraftingState tmp{};
374:     const std::uint8_t* p = in;
375:     for (std::size_t w = 0; w < kCraftKnownWords; ++w) {
376:         const std::uint32_t lo = get_u32(p);
377:         const std::uint32_t hi = get_u32(p);
378:         tmp.known[w] = static_cast<std::uint64_t>(lo) |
379:                        (static_cast<std::uint64_t>(hi) << 32);
380:     }
381:     for (std::size_t i = 0; i < kCraftMaterials; ++i) {
382:         const std::uint32_t v = get_u32(p);
383:         tmp.mat[i] = v > kCraftMaterialMax ? kCraftMaterialMax : v;
384:     }
385:     const std::uint8_t tier = *p++;
386:     tmp.tier = tier > kCraftMaxTier ? kCraftMaxTier : tier;
387: 
388:     // Drop bits naming rows the table does not have. 448 bits over 446 recipes, so
389:     // this only ever clears the top 2 of the last word today. `if constexpr`
390:     // because the count is known at compile time and MSVC /W4 raises C4127 on a
391:     // runtime `if` over a constant — and [AGENTS.md] means zero warnings.
392:     constexpr std::size_t kSpareBits = kCraftKnownWords * 64 - kCraftRecipeCount;
393:     if constexpr (kSpareBits > 0)
394:         tmp.known[kCraftKnownWords - 1] &= (~0ull >> kSpareBits);
395: 
396:     st = tmp;
397:     return true;
398: }
