1|// Key bindings — the data-driven "key -> console line" table.
2|//
3|// The last hardcoded keys lived as an `if` chain in the app's event loop; this
4|// table replaces them the same way ConsoleCommand replaced key-handler `if`s
5|// ([console.h]): a binding is a ROW (action name, scancode, console line), the
6|// app's event loop does ONE lookup per keydown and feeds the row's command to
7|// Console::exec. Keyboard, pause-menu buttons and typed console lines therefore
8|// all drive the SAME command rows — rebinding, save/load of bindings and new
9|// actions are all data edits, never app code.
10|//
11|// Pure game layer: no SDL/ImGui. Scancodes are stored as the USB HID usage
12|// codes SDL scancodes are DEFINED as (SDL_SCANCODE_A == 4, ...), so the table
13|// can name defaults and stay headless-testable — the same no-dependency stance
14|// as type_tag<T> vs RTTI. Serialization is a plain text format ("bind <action>
15|// <scancode>"); the app owns the file I/O, exactly the save.h split.
16|#pragma once

24|// Usage Tables spec (and therefore per SDL_scancode.h, verbatim).

27|    kG = 10, kH = 11, kP = 19, kQ = 20, kR = 21, kS = 22, kT = 23, kV = 25,
28|    kW = 26, kX = 27, kEscape = 41, kTab = 43, kSpace = 44, kLeftBracket = 47,

43|struct KeyBind {
44|    const char* action = "";  // stable id: the file key and the menu label
45|    const char* command = ""; // console line run on press; "" = a HELD axis key
46|                              // (polled by the input bridge, never dispatched)
47|    std::uint16_t scancode = 0;
48|    std::uint8_t flags = 0;
49|};

51|class KeybindTable {
52|public:
53|    static constexpr std::size_t kMaxBinds = 64;
54|
55|    // Register a row. Refused (false) on a duplicate action, an empty action,
56|    // or a full table — first registration wins, same rule as Console::add.
57|    bool add(const KeyBind& b);
58|
59|    const KeyBind* find(const char* action) const;
60|
61|    // The first COMMAND row bound to this scancode, or nullptr. Axis rows are
62|    // skipped, which is what lets one key be both a held axis and a pressed
63|    // command (Q descends in fly AND works a door, as it always has).
64|    const KeyBind* find_scancode(std::uint16_t sc) const;
65|
66|    // Point an existing action at a new key. Unknown action is refused.
67|    bool rebind(const char* action, std::uint16_t sc);
68|
69|    // How many OTHER command rows share this row's scancode — a real dispatch
70|    // ambiguity (only the first fires), surfaced by the menu as a conflict.
71|    std::uint32_t conflicts(const char* action) const;
72|
73|    std::size_t count() const { return count_; }
74|    const KeyBind& at(std::size_t i) const { return binds_[i]; }
75|    void clear() { count_ = 0; }
76|
77|    // One "bind <action> <scancode>" line per row. Returns chars written
78|    // (NUL-terminated, truncated at cap).
79|    std::size_t serialize(char* out, std::size_t cap) const;
80|
81|    // Apply "bind ..." lines to EXISTING rows; unknown actions and malformed
82|    // lines are skipped (a stale file must not invent actions). '#' comments.
83|    // Returns rows applied.
84|    std::uint32_t parse(const char* text);
85|
86|private:
87|    KeyBind binds_[kMaxBinds] = {};
88|    std::size_t count_ = 0;
89|};

57|    if (!out || cap == 0) return 0;
58|    std::size_t at = 0;

69|std::uint32_t KeybindTable::parse(const char* text) {
70|    if (!text) return 0;
71|    std::uint32_t applied = 0;
72|    const char* p = text;
73|    while (*p) {
74|        // One line at a time, in place — no allocation, a bad line is skipped.
75|        const char* eol = p;
76|        while (*eol && *eol != '\n') ++eol;
77|
78|        char line[128];
79|        const std::size_t len = static_cast<std::size_t>(eol - p);
80|        if (len > 0 && len + 1 < sizeof line && *p != '#') {
81|            std::memcpy(line, p, len);
82|            line[len] = '\0';
83|
84|            char action[64] = {};
85|            unsigned int sc = 0;
86|            if (std::sscanf(line, "bind %63s %u", action, &sc) == 2 &&
87|                sc <= 0xFFFFu) {
88|                if (rebind(action, static_cast<std::uint16_t>(sc))) ++applied;
89|            }
90|        }
91|        p = *eol ? eol + 1 : eol;
92|    }
93|    return applied;

1|// Debug console — the data-driven command registry behind the ~ overlay.
2|//
3|// The console is TWO pieces on the render/sim seam ([ARCHITECTURE.md] §L3):
4|//   * THIS — the registry, parser, completion and the default command set. Pure
5|//     game layer: no SDL/Vulkan/ImGui, headless-tested in game_test.
6|//   * the ImGui overlay in src/app/main.cpp — a thin shell that feeds typed
7|//     lines in and prints what comes back. Remove it and the console still
8|//     works (a dedicated-server stdin could drive the same exec()).
9|//
10|// DATA-DRIVEN, in both directions:
11|//   * a command is a ROW (name, usage, help, run, complete) — registering a
12|//     duplicate name is refused, mirroring FloorCatalog's claim rule;
13|//   * arguments complete from the live tables, not from hardcoded lists:
14|//     `spawn` enumerates kMobTokens (generated from data/mobs.csv), `teleport`
15|//     enumerates the floors actually registered in the FloorRegistry. Add a CSV
16|//     row or register a floor and the console knows it — no console edit.
17|//
18|// CLIENT PROPOSES, SERVER DISPOSES ([netcode-seam]): a command must not yank
19|// the app's world mid-frame, so cross-floor effects are REQUESTS written into
20|// the ConsoleContext (requestFloor); the app performs them at its own safe
21|// point in the frame, exactly like a PlayerCommand.

39|// One-shot actions a command may REQUEST of the app — the generic sibling of
40|// requestFloor, and the seam the keybinding table ([keybind.h]) dispatches
41|// through: a bound key, a pause-menu button and a typed console line all set
42|// the same bit, and the app performs the effect at its safe point in the
43|// frame. A bit, not a call, because most of these touch app-owned state (the
44|// pause flag, ImGui windows, the input bridge) the game layer must not see.
45|enum class ConsoleRequest : std::uint32_t {
46|    Menu,      // toggle the pause menu
47|    Quit,      // quit to desktop
48|    Hud,       // toggle the HUD overlay
49|    Console,   // toggle the console overlay
50|    Mouselook, // toggle mouse look
51|    Fly,       // toggle fly — routed through the input bridge so it stays a
52|               // PlayerCommand button ([netcode-seam]), unlike noclip
53|    FloorUp,   // ride one floor up (the `ride` command)
54|    FloorDown, // ride one floor down
55|    Save,      // save the run
56|    Load,      // load the run
57|    Heal, Eat, Drink,        // survival one-shots
58|    Door, Possess, Interact, // world interaction one-shots
59|    Sell, Vendor, Resupply,  // economy: sell haul / trader window / resupply
60|    Craft, Scrap,            // crafting window / scrap cheapest junk
61|    Count
62|};

64|constexpr std::uint32_t request_bit(ConsoleRequest r) {
65|    return 1u << static_cast<std::uint32_t>(r);
